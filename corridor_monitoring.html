} async function arcgisGenerateToken() { const portal = (process.env.ARCGIS_PORTAL_URL || 'https://www.arcgis.com').replace(/\/$/, ''); const user = (process.env.ARCGIS_USERNAME || '').trim(); const pass = (process.env.ARCGIS_PASSWORD || '').trim(); const results = json.addResults || []; for (const r of results) { if (r.success) added++; else if (r.error) errors.push({ objectId: r.objectId, error: r.error }); } } return { added, errors }; } function mergeRequestQuery(req) { const out = {}; const url = typeof req.url === 'string' ? req.url : ''; const qi = url.indexOf('?'); if (qi !== -1) { const part = url.slice(qi + 1); for (const pair of part.split('&')) { if (!pair) continue; const eq = pair.indexOf('='); '' : pair.slice(eq + 1); try { out[decodeURIComponent(k.replace(/\+/g, ' '))] = decodeURIComponent(v.replace(/\+/g, ' ')); } catch (e) { out[k] = v; } } } if (req.query && typeof req.query === 'object') Object.assign(out, req.query); return out; } function verifySyncAuth(req) { const arcgisSec = (process.env.ARCGIS_SYNC_SECRET || '').trim(); const cronSec = (process.env.CRON_SECRET || '').trim(); const h = req.headers && req.headers.authorization; if (!h || !h.startsWith('Bearer ')) { if (!arcgisSec && !cronSec) return { ok: true }; return { ok: false }; } const t = h.slice(7).trim(); if (cronSec && t === cronSec) return { ok: true }; if (arcgisSec && t === arcgisSec) return { ok: true }; if (!arcgisSec && !cronSec) return { ok: true }; return { ok: false }; } module.exports = async (req, res) => { res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type, Authorization'); if (req.method === 'OPTIONS') return res.status(200).end(); if (req.method !== 'GET' && req.method !== 'POST') return res.status(405).json({ error: 'Method not allowed' }); const auth = verifySyncAuth(req); if (!auth.ok) { return res.status(401).json({ error: 'Authorization: Bearer or (Vercel Cron).
